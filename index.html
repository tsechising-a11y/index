import React, { useState, useEffect } from 'react';

import { ChevronRight, ChevronLeft, Copy, Check, Share2, Heart, Shield, Sun, History, MessageSquareQuote } from 'lucide-react';


const App = () => {

  const [currentStep, setCurrentStep] = useState(0);

  const [data, setData] = useState({

    steps: [

      { caretaker: '', personality: '', influence: '' }, // 第一站

      { shameMoment: '' },                             // 第二站

      { unexpectedGrace: '' },                         // 第三站

      { earliestMemory: '' },                          // 第四站

      { godFootprints: '' }                            // 第五站

    ],

    finalSummary: ''

  });

  const [copied, setCopied] = useState(false);


  const stepsInfo = [

    { title: "第一站：兒時照顧者", icon: <Heart className="w-6 h-6" /> },

    { title: "第二站：羞愧時刻", icon: <Shield className="w-6 h-6" /> },

    { title: "第三站：出奇不意的恩典", icon: <Sun className="w-6 h-6" /> },

    { title: "第四站：最早的回憶", icon: <History className="w-6 h-6" /> },

    { title: "第五站：上帝介入的足跡", icon: <MessageSquareQuote className="w-6 h-6" /> }

  ];


  // Load from LocalStorage

  useEffect(() => {

    const saved = localStorage.getItem('walking_meditation_v2');

    if (saved) {

      try {

        const parsed = JSON.parse(saved);

        if (parsed.steps && parsed.steps.length === 5) {

          setData(parsed);

        }

      } catch (e) {

        console.error("Failed to load data");

      }

    }

  }, []);


  // Save to LocalStorage

  useEffect(() => {

    localStorage.setItem('walking_meditation_v2', JSON.stringify(data));

  }, [data]);


  const handleInputChange = (field, value) => {

    const newSteps = [...data.steps];

    newSteps[currentStep] = { ...newSteps[currentStep], [field]: value };

    setData({ ...data, steps: newSteps });

  };


  const exportData = () => {

    let text = `📝【散步隨筆：生命足跡紀錄】\n\n`;

    

    // 第一站

    text += `📍 第一站：兒時照顧者\n`;

    text += `▪️ 主要照顧者：${data.steps[0].caretaker || '未填寫'}\n`;

    text += `▪️ 形象特質：${data.steps[0].personality || '未填寫'}\n`;

    text += `▪️ 性情價值觀影響：${data.steps[0].influence || '未填寫'}\n\n`;


    // 第二站

    text += `📍 第二站：羞愧時刻\n`;

    text += `▪️ 經歷分享：${data.steps[1].shameMoment || '未填寫'}\n\n`;


    // 第三站

    text += `📍 第三站：出奇不意的恩典\n`;

    text += `▪️ 得到幫助的經歷：${data.steps[2].unexpectedGrace || '未填寫'}\n\n`;


    // 第四站

    text += `📍 第四站：最早的回憶\n`;

    text += `▪️ 回憶內容：${data.steps[3].earliestMemory || '未填寫'}\n\n`;


    // 第五站

    text += `📍 第五站：上帝介入的足跡\n`;

    text += `▪️ 生命看見：${data.steps[4].godFootprints || '未填寫'}\n\n`;


    text += `✨ 散步歷程中最深刻的是：\n${data.finalSummary || '未填寫'}`;


    const textArea = document.createElement("textarea");

    textArea.value = text;

    document.body.appendChild(textArea);

    textArea.select();

    try {

      document.execCommand('copy');

      setCopied(true);

      setTimeout(() => setCopied(false), 2000);

    } catch (err) {

      alert("複製失敗，請手動複製內容。");

    }

    document.body.removeChild(textArea);

  };


  const renderCurrentStepFields = () => {

    switch (currentStep) {

      case 0:

        return (

          <div className="space-y-6">

            <div className="space-y-2">

              <label className="text-lg font-semibold block">童年時期，主要照顧我的人是...</label>

              <textarea 

                className="w-full bg-white border border-[#E5E7EB] rounded-2xl p-4 min-h-[80px] focus:ring-2 focus:ring-[#C19A6B] outline-none"

                value={data.steps[0].caretaker}

                onChange={(e) => handleInputChange('caretaker', e.target.value)}

                placeholder="例如：婆婆、媽媽..."

              />

            </div>

            <div className="space-y-2">

              <label className="text-lg font-semibold block">他/她是一個怎樣的人？</label>

              <textarea 

                className="w-full bg-white border border-[#E5E7EB] rounded-2xl p-4 min-h-[100px] focus:ring-2 focus:ring-[#C19A6B] outline-none"

                value={data.steps[0].personality}

                onChange={(e) => handleInputChange('personality', e.target.value)}

                placeholder="描述他/她的性格或特質..."

              />

            </div>

            <div className="space-y-2">

              <label className="text-lg font-semibold block">他/她對我的性情、價值觀有何影響？</label>

              <textarea 

                className="w-full bg-white border border-[#E5E7EB] rounded-2xl p-4 min-h-[120px] focus:ring-2 focus:ring-[#C19A6B] outline-none"

                value={data.steps[0].influence}

                onChange={(e) => handleInputChange('influence', e.target.value)}

                placeholder="哪些習慣或想法受他/她影響？"

              />

            </div>

          </div>

        );

      case 1:

        return (

          <div className="space-y-2">

            <label className="text-lg font-semibold block">我曾經覺得自己好羞愧的時刻是...</label>

            <textarea 

              className="w-full bg-white border border-[#E5E7EB] rounded-2xl p-4 min-h-[250px] focus:ring-2 focus:ring-[#C19A6B] outline-none"

              value={data.steps[1].shameMoment}

              onChange={(e) => handleInputChange('shameMoment', e.target.value)}

              placeholder="試著描述那份尷尬或無力感..."

            />

          </div>

        );

      case 2:

        return (

          <div className="space-y-2">

            <label className="text-lg font-semibold block">那次，我曾經得到過無條件的幫助...</label>

            <textarea 

              className="w-full bg-white border border-[#E5E7EB] rounded-2xl p-4 min-h-[250px] focus:ring-2 focus:ring-[#C19A6B] outline-none"

              value={data.steps[2].unexpectedGrace}

              onChange={(e) => handleInputChange('unexpectedGrace', e.target.value)}

              placeholder="分享一個經歷恩典的故事..."

            />

          </div>

        );

      case 3:

        return (

          <div className="space-y-2">

            <label className="text-lg font-semibold block">我最早與家人相處的回憶／最深刻與家人相處的回憶是...</label>

            <textarea 

              className="w-full bg-white border border-[#E5E7EB] rounded-2xl p-4 min-h-[250px] focus:ring-2 focus:ring-[#C19A6B] outline-none"

              value={data.steps[3].earliestMemory}

              onChange={(e) => handleInputChange('earliestMemory', e.target.value)}

              placeholder="那個畫面、氣味或感覺..."

            />

          </div>

        );

      case 4:

        return (

          <div className="space-y-2">

            <label className="text-lg font-semibold block">神怎樣讓我看到祂在我生命中出現？</label>

            <textarea 

              className="w-full bg-white border border-[#E5E7EB] rounded-2xl p-4 min-h-[250px] focus:ring-2 focus:ring-[#C19A6B] outline-none"

              value={data.steps[4].godFootprints}

              onChange={(e) => handleInputChange('godFootprints', e.target.value)}

              placeholder="在人生的起跌中，哪裡有祂的蹤跡？"

            />

          </div>

        );

      default:

        return null;

    }

  };


  return (

    <div className="min-h-screen bg-[#FDFCFB] text-[#374151] pb-24">

      {/* Header */}

      <header className="sticky top-0 bg-white/80 backdrop-blur-md border-b border-[#F3F4F6] z-10 px-6 py-4">

        <div className="max-w-md mx-auto flex flex-col items-center">

          <h1 className="text-xl font-bold tracking-tight text-[#4B3621]">散步隨筆</h1>

          <div className="flex w-full gap-1 mt-4">

            {[...Array(6)].map((_, i) => (

              <div 

                key={i} 

                className={`h-1.5 flex-1 rounded-full transition-all duration-500 ${i <= currentStep ? 'bg-[#C19A6B]' : 'bg-[#E5E7EB]'}`}

              />

            ))}

          </div>

        </div>

      </header>


      <main className="max-w-md mx-auto p-6">

        {currentStep < 5 ? (

          <div className="space-y-8 animate-in fade-in slide-in-from-bottom-4">

            <div className="flex items-center gap-3 text-[#C19A6B]">

              {stepsInfo[currentStep].icon}

              <span className="text-sm font-bold tracking-widest uppercase">

                Step {currentStep + 1}

              </span>

            </div>

            

            <h2 className="text-2xl font-bold text-[#4B3621]">

              {stepsInfo[currentStep].title}

            </h2>


            {renderCurrentStepFields()}

          </div>

        ) : (

          <div className="space-y-8 animate-in fade-in zoom-in-95">

            <div className="text-center space-y-4">

              <div className="w-16 h-16 bg-[#F3EFE0] rounded-full flex items-center justify-center mx-auto">

                <Share2 className="w-8 h-8 text-[#C19A6B]" />

              </div>

              <h2 className="text-2xl font-bold text-[#4B3621]">漫步旅程告一段落</h2>

              <p className="text-gray-500">將這段路的心情，化作文字帶走。</p>

            </div>


            <div className="space-y-2">

              <label className="text-lg font-semibold block">在這次散步歷程中，我最深刻的是...</label>

              <textarea 

                className="w-full bg-white border border-[#E5E7EB] rounded-2xl p-4 min-h-[180px] focus:ring-2 focus:ring-[#C19A6B] outline-none"

                value={data.finalSummary}

                onChange={(e) => setData({...data, finalSummary: e.target.value})}

                placeholder="寫下今天最想留住的一份感覺..."

              />

            </div>


            <button 

              onClick={exportData}

              className={`w-full py-4 rounded-2xl font-bold text-lg shadow-lg flex items-center justify-center gap-2 transition-all active:scale-95 ${copied ? 'bg-green-500 text-white' : 'bg-[#4B3621] text-white'}`}

            >

              {copied ? <Check className="w-5 h-5" /> : <Copy className="w-5 h-5" />}

              {copied ? '已複製！' : '匯出成果並複製'}

            </button>

            <p className="text-center text-xs text-gray-400">複製後可直接貼到 WhatsApp 群組分享</p>

          </div>

        )}

      </main>


      {/* Footer Navigation */}

      <footer className="fixed bottom-0 left-0 right-0 bg-white/90 backdrop-blur-md border-t border-[#F3F4F6] p-4">

        <div className="max-w-md mx-auto flex gap-4">

          <button 

            disabled={currentStep === 0}

            onClick={() => setCurrentStep(prev => prev - 1)}

            className={`flex-1 py-4 rounded-2xl font-bold flex items-center justify-center gap-1 transition-all ${currentStep === 0 ? 'bg-gray-50 text-gray-300' : 'bg-gray-100 text-[#4B3621] active:bg-gray-200'}`}

          >

            <ChevronLeft className="w-5 h-5" /> 上一站

          </button>

          <button 

            disabled={currentStep === 5}

            onClick={() => setCurrentStep(prev => prev + 1)}

            className={`flex-1 py-4 rounded-2xl font-bold flex items-center justify-center gap-1 transition-all ${currentStep === 5 ? 'bg-gray-50 text-gray-300' : 'bg-[#4B3621] text-white shadow-md active:bg-[#3D2C1B]'}`}

          >

            {currentStep === 4 ? '完成散步' : '下一站'} <ChevronRight className="w-5 h-5" />

          </button>

        </div>

      </footer>

    </div>

  );

};


export default App;
